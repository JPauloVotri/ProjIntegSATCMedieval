// ======================================================
// USUÁRIO
// ======================================================

Table Usuario {
  Id int [pk, increment]

  Nome varchar(150) [not null]
  Email varchar(150) [not null, unique]
  SenhaHash varchar(255) [not null]

  GrupoUsuario GrupoUsuario [not null, default: 'administrador']
  Status UsuarioStatus [not null, default: 'ativo']

  UltimoLogin timestamp

  CriadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]
  AtualizadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    Email [unique]
    GrupoUsuario
    Status
  }
}


// ======================================================
// GRUPO PRODUTO
// ======================================================

Table GrupoProduto {
  Id int [pk, increment]

  Nome varchar(100) [not null, unique]
  Descricao varchar(255)

  Ativo boolean [not null, default: true]

  CriadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]
  AtualizadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    Nome [unique]
    Ativo
  }
}


// ======================================================
// PRODUTO
// ======================================================

Table Produto {
  Id int [pk, increment]

  GrupoProdutoId int [not null]
  UnidadeMedidaId int [not null]

  Codigo varchar(50) [unique]
  Nome varchar(150) [not null]
  Descricao varchar(255)

  EstoqueMinimo decimal(14,3) [not null, default: 0]
  EstoqueMaximo decimal(14,3)

  CustoMedioEstoque decimal(14,4) [not null, default: 0]
  UltimoCusto decimal(14,4) [not null, default: 0]

  ValidadeDias int [default: null]

  Ativo boolean [not null, default: true]

  CriadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]
  AtualizadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    Codigo [unique]
    Nome
    GrupoProdutoId
    UnidadeMedidaId
    Ativo
  }
}


// ======================================================
// UNIDADE DE MEDIDA
// ======================================================

Table UnidadeMedida {
  UnidadeMedidaId int [pk, increment]

  Nome varchar(100) [not null]
  Sigla varchar(20) [not null, unique]
  Tipo varchar(50) [not null]
  Ativo boolean [not null, default: true]
  CriadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]
  AtualizadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    Sigla [unique]
    Tipo
    Ativo
  }
}


// ======================================================
// FORNECEDOR
// ======================================================

Table Fornecedor {
  Id int [pk, increment]

  RazaoSocial varchar(200) [not null]
  NomeFantasia varchar(200)

  Documento varchar(30)
  InscricaoEstadual varchar(30)

  Email varchar(150)
  Telefone varchar(30)
  NomeContato varchar(150)

  Cep varchar(10)
  Logradouro varchar(200)
  Numero varchar(20)
  Complemento varchar(100)
  Bairro varchar(100)
  Cidade varchar(100)
  Estado varchar(2)

  DiasEntrega varchar(50)
  CondicaoPagamento varchar(150)

  Ativo boolean [not null, default: true]

  CriadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]
  AtualizadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    Documento
    RazaoSocial
    NomeFantasia
    Ativo
  }
}


// ======================================================
// COTAÇÃO
// ======================================================

Table Cotacao {
  Id int [pk, increment]

  UsuarioId int [not null]

  DataSolicitacao date [not null]
  DataLimiteResposta date

  Status StatusCotacao [not null, default: 'rascunho']

  Observacao text

  CriadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]
  AtualizadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    UsuarioId
    Status
    DataSolicitacao
    DataLimiteResposta
  }
}


// ======================================================
// ITEM DA COTAÇÃO

// ======================================================

// ======================================================

    Origem
    CriadoEm

  }

// ======================================================
// AUDITORIA
// ======================================================
/*
É o fluxo responsável por todos os registros de alterações:
Manuais (Vinculadas a um usuário)
Automáticas (Algum processamento realizado pelo sistema)
Serve também como um registro de alterações em LOG
OBS: Confirmar se será somente via banco ou via interface
*/

Table Auditoria {
  Id int [pk, increment]

  UsuarioId int

  Tabela varchar(100) [not null]
  RegistroId int

  Tipo TipoAuditoria [not null]

  DadosAnteriores json
  DadosNovos json

  Ip varchar(45)
  UserAgent varchar(500)

  CriadoEm timestamp [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    UsuarioId
    Tabela
    RegistroId
    Tipo
    CriadoEm
  }
}


// ======================================================
// RELACIONAMENTOS
// ======================================================

// ======================================================
// CATEGORIA / PRODUTO / UNIDADE
// ======================================================

// Categoria -> Produto
Ref: GrupoProduto.Id < Produto.GrupoProdutoId

// Unidade de medida -> Produto
Ref: UnidadeMedida.UnidadeMedidaId < Produto.UnidadeMedidaId


// ======================================================
// ESTOQUE
// ======================================================

// Produto -> Estoque
// 1 produto possui 1 registro de estoque
Ref: Produto.Id - Estoque.ProdutoId


// ======================================================
// COTAÇÃO
// ======================================================

// Usuário -> Cotação
Ref: Usuario.Id < Cotacao.UsuarioId

// Cotação -> Itens
Ref: Cotacao.Id < CotacaoItem.CotacaoId

// Produto -> Itens da cotação
Ref: Produto.Id < CotacaoItem.ProdutoId

// Unidade -> Itens da cotação
Ref: UnidadeMedida.UnidadeMedidaId < CotacaoItem.UnidadeMedidaId


// ======================================================
// COTAÇÃO / FORNECEDORES
// ======================================================

// Item da cotação -> Ofertas dos fornecedores
Ref: CotacaoItem.Id < CotacaoItemFornecedor.CotacaoItemId

// Fornecedor -> Ofertas dos fornecedores
Ref: Fornecedor.Id < CotacaoItemFornecedor.FornecedorId


// ======================================================
// PEDIDO DE COMPRA
// ======================================================

// Cotação -> Pedidos de compra
// Uma cotação pode gerar vários pedidos,
// normalmente um por fornecedor escolhido.
Ref: Cotacao.Id < PedidoCompra.CotacaoId

// Fornecedor -> Pedido de compra
Ref: Fornecedor.Id < PedidoCompra.FornecedorId

// Usuário -> Pedido de compra
Ref: Usuario.Id < PedidoCompra.UsuarioId

// Pedido -> Itens
Ref: PedidoCompra.Id < PedidoCompraItem.PedidoCompraId

// Cotação -> Item do pedido
Ref: CotacaoItem.Id < PedidoCompraItem.CotacaoItemId

// Produto -> Item do pedido
Ref: Produto.Id < PedidoCompraItem.ProdutoId

// Unidade -> Item do pedido
Ref: UnidadeMedida.UnidadeMedidaId < PedidoCompraItem.UnidadeMedidaId


// ======================================================
// SAÍDA DE ESTOQUE
// ======================================================

// Usuário -> Saída de estoque
Ref: Usuario.Id < SaidaEstoque.UsuarioId

// Saída -> Itens
Ref: SaidaEstoque.Id < SaidaEstoqueItem.SaidaEstoqueId

// Produto -> Itens da saída
Ref: Produto.Id < SaidaEstoqueItem.ProdutoId

// Unidade -> Itens da saída
Ref: UnidadeMedida.UnidadeMedidaId < SaidaEstoqueItem.UnidadeMedidaId


// ======================================================
// MOVIMENTAÇÃO DE ESTOQUE
// ======================================================

// Estoque -> Movimentações
Ref: Estoque.Id < MovimentacaoEstoque.EstoqueId

// Usuário -> Movimentações
Ref: Usuario.Id < MovimentacaoEstoque.UsuarioId


// ======================================================
// AUDITORIA
// ======================================================

// Usuário -> Auditoria
Ref: Usuario.Id < Auditoria.UsuarioId
